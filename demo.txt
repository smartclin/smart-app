   2 | 
   3 | export type Person = {
   4 | 	id: number;
   5 | 	firstName: string;
   6 | 	lastName: string;
   7 | 	age: number;
   8 | 	visits: number;
   9 | 	progress: number;
  10 | 	status: "relationship" | "complicated" | "single";
  11 | 	subRows?: Person[];
  12 | };
  13 | 
  14 | const range = (len: number) => {
  15 | 	const arr: number[] = [];
  16 | 	for (let i = 0; i < len; i++) {
  17 | 		arr.push(i);
  18 | 	}
  19 | 	return arr;
  20 | };
  21 | 
  22 | const newPerson = (num: number): Person => {
  23 | 	return {
  24 | 		id: num,
  25 | 		firstName: faker.person.firstName(),
  26 | 		lastName: faker.person.lastName(),
  27 | 		age: faker.number.int(40),
  28 | 		visits: faker.number.int(1000),
  29 | 		progress: faker.number.int(100),
  30 | 		status: faker.helpers.shuffle<Person["status"]>(["relationship", "complicated", "single"])[0]!
  31 | 	};
  32 | };
  33 | 
  34 | export function makeData(...lens: number[]) {
  35 | 	const makeDataLevel = (depth = 0): Person[] => {
  36 | 		const len = lens[depth]!;
  37 | 		return range(len).map((index): Person => {
  38 | 			return {
  39 | 				...newPerson(index),
  40 | 				subRows: lens[depth + 1] ? makeDataLevel(depth + 1) : undefined
  41 | 			};
  42 | 		});
  43 | 	};
  44 | 
  45 | 	return makeDataLevel();
  46 | }

 316 | // A typical debounced input react component
 317 | function DebouncedInput({
 318 | 	value: initialValue,
 319 | 	onChange,
 320 | 	debounce = 500,
 321 | 	...props
 322 | }: {
 323 | 	value: string | number;
 324 | 	onChange: (value: string | number) => void;
 325 | 	debounce?: number;
 326 | } & Omit<React.InputHTMLAttributes<HTMLInputElement>, "onChange">) {
 327 | 	const [value, setValue] = React.useState(initialValue);
 328 | 
 329 | 	React.useEffect(() => {
 330 | 		setValue(initialValue);
 331 | 	}, [initialValue]);
 332 | 
 333 | 	React.useEffect(() => {
 334 | 		const timeout = setTimeout(() => {
 335 | 			onChange(value);
 336 | 		}, debounce);
 337 | 
 338 | 		return () => clearTimeout(timeout);
 339 | 	}, [value, onChange, debounce]);
 340 | 
 341 | 	return (
 342 | 		<input
 343 | 			{...props}
 344 | 			onChange={e => setValue(e.target.value)}
 345 | 			value={value}
 346 | 		/>
 347 | 	);
